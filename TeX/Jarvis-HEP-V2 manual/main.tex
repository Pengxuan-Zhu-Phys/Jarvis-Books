%%
% Compact table of contents (tufte's default leaves a lot of vertical air between
% entries; we tighten part/chapter spacing so the chapters-only ToC stays short).
\titlecontents{part}[0pt]%
  {\addvspace{0.9\baselineskip}\large\sffamily\bfseries}%
  {\allcaps{Part~\thecontentslabel}\quad}{\allcaps{}}%
  {}[\vspace{0.15\baselineskip}]
\titlecontents{chapter}[2.6em]%
  {\itshape}%
  {\contentslabel{2.4em}}{}%
  {\qquad\normalfont\thecontentspage}[\vspace{0.12\baselineskip}]

%%
% Number chapters AND sections (tufte-book leaves both unnumbered by default).
% secnumdepth=1 numbers chapters and sections, not subsections/paragraphs.
\setcounter{secnumdepth}{1}

%%
% tufte-book's own \chapter title (tufte-common.def) uses titlesec's [display]
% shape: the numeral on its own line, the title on the next. Put them on one
% line instead, matching \section's [hang] shape below it -- same label/title
% styling as the class default, just re-shaped. Placed here (after the class
% loads titlesec internally) so it overrides the class's definition.
\titleformat{\chapter}%
  [hang]% shape -- numeral and title share one line
  {\relax\ifthenelse{\NOT\boolean{@tufte@symmetric}}{\begin{fullwidth}}{}}%
  {\itshape\huge\thechapter}% label
  {1em}% horizontal separation between label and title body
  {\huge\rmfamily\itshape}% before the title body
  [\ifthenelse{\NOT\boolean{@tufte@symmetric}}{\end{fullwidth}}{}]%

%%
% Restore a recto start (with blank-page separation) for PART dividers, while
